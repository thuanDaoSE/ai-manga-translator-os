# 1. Xóa container cũ đi cho sạch sẽ
docker rm -f deepkoma-db

# 2. Tạo lại container, đổi cổng từ 5432 thành 5433 (Nhìn kỹ đoạn -p 5433:5432)

  -e POSTGRES_USER=deepkoma \
  -e POSTGRES_PASSWORD=secretpassword \
  -e POSTGRES_DB=manga_studio \
  -p 5433:5432 \
  -d ankane/pgvector:v0.5.1
